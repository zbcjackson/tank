"""Offline replay of explicit-status screening: requests, images, parser and outcome scores."""
import gzip
import hashlib
import json
import math
from pathlib import Path
import sys
from PIL import Image, ImageChops
from openai.types.chat import ChatCompletion
from tank_backend.tools.computer_grounding import GroundingAdapter
from tank_backend.benchmarks.grounding_probe import score_location

out=Path(sys.argv[1])
manifest=json.loads((out/'manifest.json').read_text())
results=json.loads((out/'results.json').read_text())
rows=results['results']
specs={s['id']:s for s in manifest['specs']}
assert len({r['id'] for r in rows})==len(rows)
assert [r['id'] for r in rows]==[s['id'] for s in manifest['specs']][:len(rows)]
image_checks=[]
for key,layout in manifest['layouts'].items():
    sha=layout['image_sha256']
    assert hashlib.sha256((out/f'{sha}.png').read_bytes()).hexdigest()==sha
    base=manifest['layouts'][f"{layout['seed']}-found"]
    with Image.open(out/f'{sha}.png') as image, Image.open(out/f"{base['image_sha256']}.png") as original:
        assert list(image.size)==layout['size']==base['size']
        bbox=ImageChops.difference(image.convert('RGB'),original.convert('RGB')).getbbox()
        if layout['expected']=='found':
            assert bbox is None
        else:
            x,y=layout['changed_center']
            assert bbox is not None
            assert x-24<=bbox[0]<bbox[2]<=x+25 and y-14<=bbox[1]<bbox[3]<=y+15
        image_checks.append({'case_id':key,'hash_verified':True,'changed_pixels_bbox':bbox,'changes_confined_to_label':True})
checks=[]
for row in rows:
    ident=row['id']; spec=specs[ident]; layout=manifest['layouts'][row['case_id']]
    sha=layout['image_sha256']
    assert row['expected']==spec['expected']==layout['expected']
    request=json.loads((out/f'{ident}.request.json').read_text())
    assert request==json.loads((out/f'{ident}.frozen-request.json').read_text())
    assert request['messages'][1]['content'][1]['image_url']['url']=='sha256:'+sha
    assert row['http_attempts']==1
    if row.get('usage') is None:
        checks.append({'id':ident,'request_verified':True,'response_score':'unknown: no usage/SDK response'})
        continue
    response=ChatCompletion.model_validate_json(gzip.decompress((out/f'{ident}.response.json.gz').read_bytes()))
    assert response.model_dump()==json.loads((out/f'{ident}.parsed-response.json').read_text())
    assert response.model==row['returned_model']
    assert response.usage.model_dump()==row['usage']
    assert [c.model_dump() for c in response.choices]==row['choices']
    adapter=GroundingAdapter(protocol=spec['protocol'],status_field=True)
    try:
        location=adapter.parse_response(response,tuple(layout['size']))
    except ValueError as exc:
        assert not any(row[k] for k in ('schema_valid','hit','status_correct','case_success','accepted_location_on_negative'))
        assert str(exc)==row['parse_error']
    else:
        assert row['schema_valid'] and location.status==row['location_status']
        assert (list(location.point) if location.point is not None else None)==row['point']
        assert (list(location.box) if location.box is not None else None)==row['box']
        if location.point is not None:
            score=score_location(location.point,tuple(layout['bounds']),layout['radius'])
            assert score['hit']==row['hit']
            assert math.isclose(score['distance'],row['distance'],rel_tol=0,abs_tol=1e-10)
        else:
            assert not row['hit']
        correct=location.status==layout['expected']
        assert row['status_correct']==correct
        assert row['accepted_location_on_negative']==(layout['expected']!='found' and location.point is not None)
        assert row['case_success']==(correct and (row['hit'] if layout['expected']=='found' else location.point is None))
    checks.append({'id':ident,'request_verified':True,'response_and_score_verified':True})
groups=[]
for alias in dict.fromkeys(s['alias'] for s in manifest['specs']):
    for expected in ('found','not_found','ambiguous'):
        subset=[r for r in rows if r['alias']==alias and r['expected']==expected]
        statuses={status:sum(r.get('location_status')==status for r in subset) for status in ('found','not_found','ambiguous')}
        groups.append({'alias':alias,'expected':expected,'attempts':len(subset),
            'schema_valid':sum(r['schema_valid'] for r in subset),
            'status_correct':sum(r['status_correct'] for r in subset),
            'case_success':sum(r['case_success'] for r in subset),
            'positive_hits':sum(r['hit'] for r in subset) if expected=='found' else None,
            'accepted_location_on_negative':sum(r['accepted_location_on_negative'] for r in subset),
            'returned_status_counts':statuses,
            'truncations':sum(any(c['finish_reason']=='length' for c in r.get('choices',[])) for r in subset),
            'known_tokens':sum(r['usage']['total_tokens'] for r in subset if r.get('usage')),
            'unknown_usage':sum(r.get('usage') is None for r in subset),
            'returned_models':sorted({r['returned_model'] for r in subset if 'returned_model' in r})})
known=sum(r['usage']['total_tokens'] for r in rows if r.get('usage'))
assert known==results['known_batch_tokens']
assert known+319528==results['known_screening_tokens']
assert known+319528+26000==results['accounted_screening_tokens']
summary={'requests':len(rows),'http_attempts':sum(r['http_attempts'] for r in rows),
         'independent_layouts':3,'condition_images':9,'repetitions_per_setting':1,
         'stage_requests_remaining':144-105-len(rows),'cumulative_screening_requests':105+len(rows),
         'known_batch_tokens':known,'known_screening_tokens':known+319528,
         'prior_unknown_usage_reservation':26000,'accounted_screening_tokens':known+345528,
         'remaining_tokens_after_reservation':1000000-known-345528,
         'unknown_usage':sum(r.get('usage') is None for r in rows),
         'stop_reason':results['stop_reason'],'groups':groups,
         'usage_based_uncached_peak_estimate_usd':sum((r['usage']['prompt_tokens']*specs[r['id']]['rates'][0]+r['usage']['completion_tokens']*specs[r['id']]['rates'][1])/1000000 for r in rows if r.get('usage')),
         'provider_reported_cost_usd':None,'actual_total_billed_cost':'unknown: direct providers do not return cost',
         'preflight_totals_unchanged':manifest['completed_preflight'],
         'holdout_calls':0,'desktop_actions':0,'real_screenshots_sent':0}
for name,data in [('offline-replay.json',{'checks':checks,'requests':len(checks),'image_checks':image_checks}),('summary.json',summary)]:
    (out/name).write_text(json.dumps(data,indent=2)+'\n')
(out/'replay.txt').write_bytes(Path(__file__).read_bytes())
print(json.dumps(summary,indent=2))
