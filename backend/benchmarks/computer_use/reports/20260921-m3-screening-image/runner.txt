def save(path, data):
    path.write_text(json.dumps(data, indent=2)+'\n')

def sanitized(request, digest):
    body = json.loads(request.content)
    images = []
    for msg in body['messages']:
        if not isinstance(msg['content'], list):
            continue
        for part in msg['content']:
            if part['type'] == 'image_url':
                image = base64.b64decode(part['image_url']['url'].split(',', 1)[1], validate=True)
                sha = hashlib.sha256(image).hexdigest()
                images.append(sha)
                part['image_url']['url'] = 'sha256:' + sha
    assert images == [digest], 'outbound image mismatch'
    return body

async def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--live', action='store_true')
    args = parser.parse_args()
    out = args.output
    cases = {seed:make_case(seed, 'point') for seed in SEEDS}
    for case in cases.values():
        # Exact 2x client-side replication. Score by the known inverse transform.
        original=Image.open(io.BytesIO(case['png']))
        enlarged=original.resize((original.width*2,original.height*2),Image.Resampling.NEAREST)
        assert enlarged.resize(original.size,Image.Resampling.NEAREST).tobytes()==original.tobytes()
        buf=io.BytesIO(); enlarged.save(buf,format='PNG')
        case['png']=buf.getvalue()
        case['original_size']=case['size']
        case['size']=enlarged.size
    observations = {seed:Observation.capture(case['png'], session_id='m3-image-screening', display_id=1)
                    for seed,case in cases.items()}
    layouts = {str(seed):dict(seed=seed,image_sha256=observations[seed][0].image_sha256,
                **{key:case[key] for key in ('size','original_size','target','bounds','radius','center')})
               for seed,case in cases.items()}
    holdout_hashes = {r['image_sha256'] for r in json.loads(HOLDOUT.read_text())}
    assert len({r['image_sha256'] for r in layouts.values()}) == 4
    assert not holdout_hashes.intersection(r['image_sha256'] for r in layouts.values())
    prior = json.loads((PRIOR/'summary.json').read_text())
    assert prior['cumulative_screening_requests']==73 and prior['known_screening_tokens']==150614
    assert prior['prior_unknown_usage_reservation']==26000 and prior['new_unknown_usage']==0
    for layout in layouts.values():
        assert layout['size'][0]*layout['size'][1]>2621440
        old=make_case(layout['seed'],'point')['png']
        assert (ORIGINAL/(hashlib.sha256(old).hexdigest()+'.png')).read_bytes()==old
    specs = []
    for i, seed in enumerate(SEEDS):
        models = ALIASES[i:]+ALIASES[:i]
        for alias, model, rates in models:
            for arm in (['control','treatment'] if i%2==0 else ['treatment','control']):
                thinking=alias in ('qwen38flash','deepseek')
                provider = 'deepseek' if alias=='deepseek' else 'qwen'
                specs.append({'id': f'{alias}-{seed}-{arm}', 'alias':alias,'model':model,
                    'rates':rates,'provider':provider,'seed':seed,'protocol':'point','thinking':thinking,'arm':arm,
                    'detail':'low' if alias=='deepseek' and arm=='treatment' else 'auto',
                    'endpoint': {'deepseek':'https://api.deepseek.com/beta',
                        'qwen':'https://dashscope.aliyuncs.com/compatible-mode/v1'}[provider],
                    'temperature':.1,
                    'extra_body': {'deepseek':{'thinking':{'type':'enabled' if thinking else 'disabled'}},
                        'qwen':{'enable_thinking':thinking,'vl_high_resolution_images':arm=='treatment'}}[provider]})
    if args.live:
        manifest = json.loads((out/'manifest.json').read_text())
        assert manifest['specs'] == specs
        assert hashlib.sha256((PRIOR/'summary.json').read_bytes()).hexdigest()==manifest['prior_summary_sha256']
        assert manifest['layouts'] == json.loads(json.dumps(layouts))
        assert (out/'runner.txt').read_bytes() == Path(__file__).read_bytes()
        for name,digest in manifest['source_sha256'].items():
            assert hashlib.sha256((ROOT/name).read_bytes()).hexdigest() == digest
        assert hashlib.sha256(HOLDOUT.read_bytes()).hexdigest() == manifest['holdout_input_sha256']
        load_dotenv(ROOT/'core/.env')
        qwen = AppConfig.load(ROOT/'core/config.yaml').get_llm_profile('computer_use')
        assert qwen.base_url.rstrip('/') == 'https://dashscope.aliyuncs.com/compatible-mode/v1'
        assert qwen.model == ALIASES[0][1]
        keys = {'qwen':qwen.api_key,'deepseek':os.environ.get('DEEPSEEK_API_KEY','')}
        assert all(keys.values()), 'missing credential'
        with (out/'live-started.json').open('x') as f:
            json.dump({'started_unix':time.time()},f)
    else:
        out.mkdir(parents=True, exist_ok=False)
        (out/'runner.txt').write_bytes(Path(__file__).read_bytes())
        for observation,png in observations.values():
            (out/f'{observation.image_sha256}.png').write_bytes(png)
        manifest = {'revision':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),
            'date':'2026-09-21','scope':'M3 image screening: Qwen high-resolution false/true; DeepSeek detail auto/low',
            'specs':specs,'layouts':layouts,'previous_screening_requests':73,'requests_max':32,
            'prior_summary_sha256':hashlib.sha256((PRIOR/'summary.json').read_bytes()).hexdigest(),
            'stage_requests_max':144,'previous_screening_tokens':150614,
            'prior_unknown_usage_reservation':26000,'previous_accounted_tokens':176614,'total_tokens_stop':1000000,
            'completed_preflight':{'requests':16,'tokens':26398},
            'image_factor':'Qwen vl_high_resolution_images false/true or DeepSeek image_url.detail auto/low',
            'client_resize':'2x nearest for both arms; score known inverse point/2 in original layout pixels',
            'input_geometry':'5.18-7.06 megapixels, all exceed Qwen default 2621440-pixel threshold',
            'provider_caveat':'DeepSeek thinking ignores temperature; default thinking effort high' ,
            'order':'models rotated per layout; each model has two control-first and two treatment-first pairs',
            'independent_layouts':4,'repetitions_per_layout_setting':1,
            'strict':False,'detail':'provider-specific paired settings','nullable_style':'integer','status_field':False,
            'thinking':'fixed per model: Qwen3.7/Max off, Qwen3.8 Flash/DeepSeek on','max_output_tokens_per_call':16000,'per_call_seconds':90,
            'batch_seconds_stop':3600,'retries':0,'input_token_reservation_per_call':10000,
            'input_reservation_is_estimate_not_exact_tokenization':True,
            'stop':'API/timeout/missing usage/over-reservation stops batch; parser failures with usage retained',
            'estimated_cost_usd':round(sum(a*.01+b*.016 for _,_,(a,b) in ALIASES)*8,6),
            'cost_basis':'uncached peak list prices; billed cost unknown without provider usage.cost',
            'price_sources_checked_2026_09_21':SOURCES,
            'source_sha256':{name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest() for name in SOURCE},
            'holdout_input_sha256':hashlib.sha256(HOLDOUT.read_bytes()).hexdigest(),
            'holdout_image_hash_overlap':0,'holdout_calls':0,'desktop_actions':0,'real_screenshots_sent':0}
        save(out/'manifest.json',manifest)
        keys = dict.fromkeys(['qwen','deepseek'],'test')
    rows=[]
    total=150614
    stop=None
    batch_start=time.monotonic()
    for spec in specs:
        if total+26000+26000>1000000 or time.monotonic()-batch_start>=3600 or len(rows)>=32:
            stop='token, time or request budget exhausted'
            break
        case=cases[spec['seed']]
        observation,png=observations[spec['seed']]
        row={'id':spec['id'],'alias':spec['alias'],'seed':spec['seed'],'protocol':spec['protocol'],'thinking':spec['thinking'],'arm':spec['arm'],'detail':spec['detail'],
             'image_sha256':observation.image_sha256,'schema_valid':False,'hit':False,'http_attempts':0}
        adapter=GroundingAdapter(protocol=spec['protocol'],detail=spec['detail'])
        profile=LLMProfile(name=spec['id'],api_key=keys[spec['provider']],model=spec['model'],
            base_url=spec['endpoint'],temperature=spec['temperature'],max_tokens=16000,
            extra_body=spec['extra_body'])
        llm=create_llm_from_profile(profile)
        await llm.client.close()
        async def request_hook(request):
            body=sanitized(request,observation.image_sha256)
            assert str(request.url)==spec['endpoint'].rstrip('/')+'/chat/completions'
            if args.live:
                assert body==json.loads((out/(spec['id']+'.frozen-request.json')).read_text())
                row['http_attempts']+=1
                save(out/(spec['id']+'.request.json'),body)
            else:
                save(out/(spec['id']+'.frozen-request.json'),body)
        async def response_hook(response):
            raw=await response.aread()
            if args.live:
                (out/(spec['id']+'.response.json.gz')).write_bytes(gzip.compress(raw))
                row['http_status']=response.status_code
                row['request_id']=response.headers.get('x-request-id')
        def fake(request):
            arguments={'found':True,'x':500,'y':500}
            return httpx.Response(200,json={'id':'dry','object':'chat.completion','created':1,
                'model':spec['model'],'choices':[{'index':0,'finish_reason':'tool_calls','message':{
                'role':'assistant','tool_calls':[{'id':'one','type':'function','function':{
                'name':'click','arguments':json.dumps(arguments)}}]}}],
                'usage':{'prompt_tokens':100,'completion_tokens':20,'total_tokens':120}})
        http=httpx.AsyncClient(transport=None if args.live else httpx.MockTransport(fake),
                              event_hooks={'request':[request_hook],'response':[response_hook]})
        started=time.monotonic()
        async with AsyncOpenAI(api_key=profile.api_key,base_url=profile.base_url,max_retries=0,
                               timeout=85,http_client=http) as client:
            llm.client=client
            try:
                async with asyncio.timeout(min(90,3600-(time.monotonic()-batch_start))):
                    response=await adapter.request(llm,observation,png,case['target'])
                row.update(returned_model=response.model,returned_provider=(response.model_extra or {}).get('provider'),
                    usage=response.usage.model_dump() if response.usage else None,
                    choices=[c.model_dump() for c in response.choices])
                if args.live:
                    save(out/(spec['id']+'.parsed-response.json'),response.model_dump())
                try:
                    location=adapter.parse_response(response,observation.image_size)
                    row.update(schema_valid=True,location_status=location.status,point=location.point,box=location.box)
                    if location.point is not None:
                        score_point=(location.point[0]/2,location.point[1]/2)
                        row.update(score_point=score_point,**score_location(score_point,case['bounds'],case['radius']))
                except ValueError as exc:
                    row['parse_error']=str(exc)
                if response.usage is None:
                    stop='missing usage'
                else:
                    total+=response.usage.total_tokens
                    if response.usage.prompt_tokens>10000 or total+26000>1000000:
                        stop='input reservation or token budget exceeded'
            except (APIError,TimeoutError) as exc:
                row['error']=type(exc).__name__
                row['usage']=None
                stop='request failure; usage unknown'
        row['seconds']=round(time.monotonic()-started,3)
        rows.append(row)
        if args.live:
            save(out/'results.json',{'results':rows,'stop_reason':stop,'known_screening_tokens':total,'known_batch_tokens':total-150614,'accounted_screening_tokens':total+26000})
            print(json.dumps({k:row[k] for k in ('id','schema_valid','hit','seconds','usage')}),flush=True)
        if stop:
            break
    if not args.live:
        pairs=0
        for seed in SEEDS:
            for alias,_,_ in ALIASES:
                control=json.loads((out/f'{alias}-{seed}-control.frozen-request.json').read_text())
                treatment=json.loads((out/f'{alias}-{seed}-treatment.frozen-request.json').read_text())
                if alias=='deepseek':
                    assert control['messages'][1]['content'][1]['image_url'].pop('detail')=='auto'
                    assert treatment['messages'][1]['content'][1]['image_url'].pop('detail')=='low'
                else:
                    assert control.pop('vl_high_resolution_images') is False
                    assert treatment.pop('vl_high_resolution_images') is True
                assert control==treatment
                pairs+=1
        assert len(rows)==32 and all(r['schema_valid'] for r in rows)
        save(out/'offline-check.json',{'fake_http_calls':len(rows),'image_parameter_only_pairs':pairs,
             'holdout_hash_overlap':0,'all_valid_fake_responses':True,'model_calls':0})
    print('DONE',len(rows),stop,flush=True)

asyncio.run(main())
